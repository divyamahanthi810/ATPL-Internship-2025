4.Acces private and protected:
class Person {
  public name: string;
  private age: number;
  protected gender: string;
  constructor(name: string, age: number, gender: string) {
    this.name = name;
    this.age = age;
    this.gender = gender;
  }
  public getAge(): number {
    return this.age;
  }
  class Student extends Person {
  public getGender(): string {
    return this.gender; 
  }
}
const person1 = new Person("Alice", 25, "female");
console.log(person1.name);     
console.log(person1.getAge());   
const student = new Student("Bob", 20, "male");
console.log(student.getGender());

5.To check wheather same elements
function arraysEqual(firstArray: number[], secondArray: number[]): boolean {
  if (firstArray.length !== secondArray.length) 
	return false;
  const sorted1 = [...firstArray].sort();
  const sorted2 = [...secondArray].sort();
  return sorted1.every((value, index) => value === sorted2[index]);
}

function removeDuplicates(arr: number[]): number[] {
  const unique: number[] = [];
  for (let i = 0; i < arr.length; i++) {
    if (!unique.includes(arr[i])) {
      unique.push(arr[i]);
    }
  }
  return unique;
}
